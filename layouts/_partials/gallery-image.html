{{- $page := . -}} {{- $image := dict -}} {{- with .Params.image -}} {{- $path := path.Base . -}} {{- with $page.Resources.GetMatch $path -}} {{- $resource := . -}} {{- $thumb := $resource -}} {{- $full := $resource -}} {{- $srcset := "" -}} {{- $width := "" -}} {{- $height := "" -}} {{- if reflect.IsImageResourceProcessable $resource -}} {{- $thumb = $resource.Resize "800x webp q80" -}} {{- $full = $resource.Resize "1600x webp q84" -}} {{- $width = $thumb.Width -}} {{- $height = $thumb.Height -}} {{- $srcsetParts := slice -}} {{- range $candidateWidth := slice 360 540 -}} {{- if le $candidateWidth $resource.Width -}} {{- $variant := $resource.Resize (printf "%dx webp q78" $candidateWidth) -}} {{- $srcsetParts = $srcsetParts | append (printf "%s %dw" $variant.RelPermalink $variant.Width) -}} {{- end -}} {{- end -}} {{- $srcsetParts = $srcsetParts | append (printf "%s %dw" $thumb.RelPermalink $thumb.Width) | uniq -}} {{- $srcset = delimit $srcsetParts ", " -}} {{- end -}} {{- $image = dict "thumb" $thumb.RelPermalink "full" $full.RelPermalink "srcset" $srcset "width" $width "height" $height "alt" $page.Title -}} {{- else -}} {{- $url := . -}} {{- if not (or (hasPrefix $url "http://") (hasPrefix $url "https://") (hasPrefix $url "/")) -}} {{- $url = $url | relURL -}} {{- end -}} {{- $image = dict "thumb" $url "full" $url "srcset" "" "alt" $page.Title -}} {{- end -}} {{- end -}} {{- return $image -}}
